Database Design Document
    <=> we are using Local-First database along with Bring-Your-Own-Storage (BYOS) approach for our application.
    <=> Database:
        => WatermelonDB (Local-First Database) - uses indexedDB / SQLite as local storage
        => PostgreSQL (Cloud Storage Database)
        => NATS KV (In-memory Database for Caching and Real-time Data Handling)

* Cloud Storage Database Design
--------------------------------
1. USER AND DEVICE MANAGEMENT
    * Users:
        => id: "UUID, Primary Key, Immutable"

        //-- Public Identifiers
        => username: "String, Unique, Indexed, Immutable"
        => mobile: "String, Unique, Indexed, NOT NULL"
        => email: "String, Unique, Indexed, nullable until verified"

        //-- Authentication
        => password_hash: "String, internal use only (NEVER exposed to sync payloads)"

        //-- Profile Information (KYC Fields: Modifying these resets 'is_verified' to false)
        => name: "String, nullable"
        => dob: "Date, nullable"
        => gender: "Enum ('male', 'female', 'other'), nullable"
        => country: "String (ISO Country Code), nullable"

        => is_verified: "Boolean, default = false, resets to false if KYC fields are altered"

        //-- Non-KYC Profile Information (Freely editable)
        => dp: "String (URL to media storage), nullable"
        => bio: "String, nullable"

        //-- Tombstoning
        => is_deleted: "Boolean, default = false"
        => deleted_at: "Timestamp, nullable"

        //-- Audit & Local-First Sync
        => created_at: "Timestamp, Immutable"
        => updated_at: "Timestamp (used for server-side audit logs only)"
        => hlc: "String (Hybrid Logical Clock), used exclusively for calculating sync deltas"

    * UserSettings:
        => id: "UUID, Primary Key, Immutable"
        => user_id: "UUID, Foreign Key to UserId, UNIQUE, Indexed"

        //-- Localization & Status
        => language: "String (e.g., 'en', 'hi'), User's preferred language"
        => timezone: "String (e.g., 'UTC'), User's preferred timezone"

        //-- Conversation Pinning, Labels(conversation grouping), Circles(contacts grouping)
        => pinned_conversation_id: "UUID, Foreign Key to Conversations.id, nullable"
        => labels: Array<{ id: "UUID, NOT NULL", name: "String, nullable", position: "Integer, NOT NULL" }> max length = 8
        => circles: Array<{ id: "UUID, NOT NULL", name: "String, nullable", position: "Integer, NOT NULL" }> max length = 8

        //-- Account Status and Security
        => account: {
            => account_status: "Enum ('active', 'deactivated', 'deleted')"
            => two_factor_auth: "Boolean, default = false"
            => mfa: {
                sms: "String, nullable until verified",
                email: "String, nullable until verified",
                totp: "String, nullable until verified",
            }
            => recovery_codes_hash: "Array<String>, (Hash of valid 2FA recovery codes for emergency access)"
            => last_password_change: "Timestamp, nullable (Used to force-expire sessions if the user changes their password)"
        }

        //-- Notification Settings
        => notifications: {
            => email: "Boolean, Global setting for account activity"
        }

        //-- Backup Settings (BYOS Strategy)
        => backup: {
            => enabled: "Boolean"
            => provider: "String, e.g. 'S3', 'Dropbox', 'Google Drive', 'OneDrive', 'iCloud', 'Custom'",
            => backup_location: "String, e.g. 's3://bucket-name/folder', 'dropbox://folder', 'gdrive://folder', 'onedrive://folder', 'icloud://folder', 'custom://folder'",
            => backup_frequency: "String, e.g. 'daily', 'weekly', 'monthly'",
            => over_wifi_only: "Boolean, whether to only backup over Wi-Fi",
            => last_backup_at: "Timestamp, nullable (The last time the backup was performed)"
        }

        //-- Privacy Settings
        => privacy: {
            => read_receipts: "Boolean, 'if true, send read receipts to contacts'"
            // -- standard values: 'everyone', 'contacts', 'nobody'
            => last_seen: "Enum, 'everyone', 'contacts', 'nobody'"
            => email: "Enum ('everyone', 'contacts', 'nobody')"
            => dp: "Enum ('everyone', 'contacts', 'nobody')"
            => dob: "Enum ('everyone', 'contacts', 'nobody')"
            => bio: "Enum ('everyone', 'contacts', 'nobody')"
        }

        //-- Status & Audit
        => last_active_at: "Timestamp, used for presence/session tracking"
        => created_at: "Timestamp, immutable"
        => updated_at: "Timestamp Immutable"
        => hlc: "String (Hybrid Logical Clock)"

    * Device:

        //-- Audit & Sync
        => created_at: "Timestamp, Immutable"
        => updated_at: "Timestamp"
        => hlc: "String (Hybrid Logical Clock), used for sync tracking"

    * UserRelationships:
        // require users device permission to access users contacts.
        // scenario 1: if user does not allows access to their contacts.
        //      -> check if is there any contacts has been crean though their other devices.
        //          -> if Yes
        //              -> contacts tab will shows the list of contacts created through other devices.
        //              -> privacy settings will provide contacts as options along with nobody and everyone.
        //              ->  Contact privacy filter will work with contacts created through other devices, not the actual contacts in current device(which are not synced).
        //          -> if no
        //              -> the relationship is not created and the user cannot access the contacts.
        //              -> privacy settings where contacts options are there will be disabled. only nobody/everyone privacy options will be there.
        // scenario 2: if user allows access to their contacts.
        //      -> client application will create user relationship entries for those contacts from the phonebook which are a registered user on dakiya platform.
        //      -> now since we have contacts list all the features will be working as expected.

        => id: "UUID, Primary Key, Immutable"

        //-- Relationship metadata
        => owner_id: "UUID, Foreign Key to Users.id, Indexed, NOT NULL, Who created the contact relationship"
        => user_id: "UUID, Foreign Key to Users.id, Indexed, NOT NULL, Who is the contact or user"

        //-- Relationship status
        => is_contact: "Boolean, default = false"
        => is_favorite: "Boolean, default = false"
        => is_blocked: "Boolean, default = false"

        //-- Contact grouping
        => circle_ids: "Array of UUIDs, Indexed, NOT NULL, Default = [], Max allowed ids = 4"

        //-- Sync Tracking
        => created_at: "Timestamp, Immutable"
        => updated_at: "Timestamp"
        => hlc: "String (Hybrid Logical Clock)"

2. CONVERSATION MANAGEMENT
    * Conversation:
        => id: "UUID, Primary Key, Immutable"

        //-- Conversation Type
        => type: "Enum ('direct', 'group', 'channel', 'system', 'self'), NOT NULL"

        //-- Shared Metadata based on conversation type
        => metadata<T extends 'group', 'channel', 'system'>: {
            // Group Metadata, when type = 'group'
            => title: "String, NOT NULL",
            => description: "String, nullable",
            => avatar_url: "String, nullable",
            => created_by_id: "UUID, NOT NULL",
            => allow_invites: "Boolean, NOT NULL, DEFAULT false",
            => admin_only_edit_info: "Boolean, NOT NULL, DEFAULT true",
            => share_history_on_join: "Boolean, NOT NULL, DEFAULT false"

            // Channel Metadata, when type = 'channel'
            => title: "String, NOT NULL",
            => description: "String, nullable",
            => avatar_url: "String, nullable",
            => created_by_id: "UUID, NOT NULL",
            => handle: "String, nullable, Unique",
            => admin_only_messages: "Boolean, NOT NULL, DEFAULT true"

            // System Metadata, when type = 'system'
            => title: "String, NOT NULL",
            => description: "String, nullable",
            => avatar_url: "String, nullable"
        }

        //-- message pinning
        => pinned_message_root_id: "UUID, nullable"

        //-- Tombstoning (For dropping/archiving whole groups)
        => is_deleted: "Boolean, default = false"
        => deleted_at: "Timestamp, nullable"

        //-- Audit & Sync
        => created_at: "Timestamp, Immutable"
        => updated_at: "Timestamp"
        => hlc: "String (Hybrid Logical Clock), used for sync tracking"

    * ConversationMembers:
        => id: "UUID, Primary Key, Immutable"

        //-- Relational Links
        => conversation_id: "UUID, Foreign Key to Conversations.id, Indexed"
        => user_id: "UUID, Foreign Key to Users.id, Indexed"

        //-- Unique Constraint Exception
        // Must enforce a UNIQUE index on (conversation_id, user_id)
        // to prevent duplicate memberships.

        //-- Roles & Permissions
        => role: "Enum ('member', 'admin', 'owner'), default = 'member'" if group/channel conversation created creator becomes owner by default
        // Action	                        Member	    Admin	   Owner
        // Send messages & invite           Yes	        Yes	       Yes
        // Delete others' messages     	    No	        Yes	       Yes
        // Kick/ban regular members         No	        Yes	       Yes
        // Promote a member to Admin	    No	        No	       Yes
        // Demote or kick an Admin	        No	        No	       Yes
        // Transfer Ownership	            No	        No	       Yes
        // Delete the Group completely	    No	        No	       Yes
        // Rules:
        //     -> there can be multiple admins and single owners in a group/channel conversation
        //     -> if owner leaves or owner account is deleted:
        //          -> either owner will transfer ownership to another admin. if not:
        //          -> admin who has been in the group/channel the longest will be promoted to owner.
        //          -> if there is no admin:
        //              -> the member who has been in the group the longest will be promoted to owner. if no member is there the group will be deleted.
        //              -> for channels, if there is no admin, the channel will be deleted. no member can be promoted to admin/owner automatically for channels.

        => chat_prefs: {
            theme: String nullable,
            notification_sound: String nullable
        },

        //-- Notification and message pinning and labeling
        => mute_until: "Timestamp, nullable (Holds the timestamp until which notifications are muted),  past timestamp means notifications are not muted"
        => pinned_message_root_id: "UUID, nullable (Holds the root message ID for pinned messages)"
        => label_ids: "Array<UUID>, NOT NULL, default = [] (Holds the IDs of the labels applied to the message), max length = 4 (max number of labels per conversation)"

        //-- Lifecycle States and Visibility (The Tombstone Pattern for Membership and The High-Water Mark)
        => is_active: "Boolean, default = true (Set to false if user leaves or is kicked)"
        => joined_at: "Timestamp, NOT NULL"
        => left_at: "Timestamp, nullable (Populated when is_active becomes false)"
        => cleared_at: "Timestamp, nullable (Hides all messages older than this timestamp for this specific user)"

        //-- Unread Tracking (Sync Optimization)
        => last_read_message_id: "UUID, Foreign Key to Messages.id, nullable"
        => last_read_at: "Timestamp, nullable"

        //-- Audit & Sync
        => created_at: "Timestamp, Immutable"
        => updated_at: "Timestamp"
        => hlc: "String (Hybrid Logical Clock), used for sync tracking"

3. MESSAGE MANAGEMENT
    * Message:
        => id: "UUID, Primary Key, Immutable" (Generated locally by WatermelonDB so users can message offline)
        => root_id: "UUID, Immutable, Indexed, NOT NULL (The constant ID for the whole "conversation thread" of this message)"
        => version: "Integer, Immutable, NOT NULL, Default: 0 (The version number of this message. 0-indexed: 0, 1, 2...)"
        //-- Unique Constraint Exception
        // Must enforce a UNIQUE index on (root_id, version)
        // to prevent duplicate version numbers within the same root_id.

        //-- Relational Links
        => conversation_id: "UUID, Foreign Key to Conversations.id, Indexed, NOT NULL"
        => sender_id: "UUID, Foreign Key to Users.id, Indexed, nullable"
           // (Nullable strictly for auto-generated system messages, e.g., 'Alice joined the group')
        => reply_to_message_id: "UUID, Foreign Key to Message.id, Indexed, nullable"
           // (Nullable for messages that are not replies to other messages). Snapshot pointer — the exact version that existed when the reply was sent.

        => type: "Enum ('text', 'image', 'video', 'audio', 'document', 'contact', 'location', 'poll', 'event', 'system', 'delete'), NOT NULL"
            // once we get a message with delete type in messages chain for perticular root_id. there can not be any new message in that chain.

        => content<T extends 'text' | 'image' | 'video' | 'audio' | 'document' | 'contact' | 'location' | 'poll' | 'event' | 'system' | 'delete'>: {
            //-- Text Message Content
            => text: "'Text content of the message', NOT NULL for type: 'text'"

            //-- Image Message Content
            => file_path: "'Relative file path of the image', NOT NULL for type: 'image'"
            => blur_hash: "'Blur hash of the image', nullable for type: 'image'"
            => size: "Integer, size of the image in bytes, NOT NULL for type: 'image'"
            => width: "Integer, width of the image in pixels, NOT NULL for type: 'image'"
            => height: "Integer, height of the image in pixels, NOT NULL for type: 'image'"
            => caption: "'Caption for the image', nullable for type: 'image'"

            //-- Video Message Content
            => file_path: "'Relative file path of the video', NOT NULL for type: 'video'"
            => thumbnail_path: "'Relative file path of the thumbnail', nullable for type: 'video'"
            => blur_hash: "'Blur hash of the thumbnail', nullable for type: 'video'"
            => size: "Integer, size of the video in bytes, NOT NULL for type: 'video'"
            => width: "Integer, width of the video in pixels, NOT NULL for type: 'video'"
            => height: "Integer, height of the video in pixels, NOT NULL for type: 'video'"
            => duration: "Integer, duration of the video in seconds, NOT NULL for type: 'video'"
            => caption: "'Caption for the video', nullable for type: 'video'"

            //-- Audio Message Content
            => file_path: "'Relative file path of the audio', NOT NULL for type: 'audio'"
            => size: "Integer, size of the audio in bytes, NOT NULL for type: 'audio'"
            => duration: "Integer, duration of the audio in seconds, NOT NULL for type: 'audio'"
            => caption: "'Caption for the audio', nullable for type: 'audio'"

            //-- Document Message Content
            => file_path: "'Relative file path of the document', NOT NULL for type: 'document'"
            => mime_type: "'MIME type of the document', nullable for type: 'document'"
            => size: "Integer, size of the document in bytes, NOT NULL for type: 'document'"
            => caption: "'Caption for the document', nullable for type: 'document'"

            //-- Contact Message Content
            => name: "'Name of the contact', nullable for type: 'contact'"
            => phone: "'Phone number of the contact', nullable for type: 'contact'"
            => email: "'Email address of the contact', nullable for type: 'contact'"
            => address: "'Address of the contact', nullable for type: 'contact'"
            => vcard: "'VCard of the contact', nullable for type: 'contact'"

            //-- Location Message Content
            => latitude: "Float, latitude of the location, NOT NULL for type: 'location'"
            => longitude: "Float, longitude of the location, NOT NULL for type: 'location'"
            => location_name: "'Name or description of the location', nullable for type: 'location'"

            //-- Poll Message Content
            => poll_question: "'Question of the poll', NOT NULL for type: 'poll'"
            => poll_options: "Array of strings, options for the poll, NOT NULL for type: 'poll'"
            => poll_multiple_answers: "Boolean, if multiple answers are allowed, NOT NULL for type: 'poll'"
            => poll_expires_at: "'Timestamp when the poll expires', nullable for type: 'poll'"

            //-- Event Message Content
            => event_title: "'Title of the event', NOT NULL for type: 'event'"
            => event_description: "'Description of the event', nullable for type: 'event'"
            => event_location: "'Location of the event', nullable for type: 'event'"
            => event_start_time: "'Timestamp when the event starts', NOT NULL for type: 'event'"
            => event_end_time: "'Timestamp when the event ends', nullable for type: 'event'"

            //-- System Message Content
            => event_key: "String, NOT NULL, (e.g., 'member_removed', 'group_name_changed')"
            => actor_id: "UUID, nullable (If null, the 'System' did it)"
            => target_id: "UUID, nullable (Can be a User ID, Group ID, or null)"
            => value: "String, nullable (Text payload, URL, or Name)"

            //-- Delete Message Content - this will behave like delete the message for all.
            => deleted_by: "UUID, NOT NULL, user who performed the deletion, can be message creator or admin or owner of the group/channel based on user roles and permissions table"
            => reason_for_deletion: "String, nullable (Reason for deletion, if applicable)"
        }

        //-- The Grouping Engine
        => message_group_id: UUID, nullable  -- (All 4 messages in the batch share this exact same UUID)
        => message_group_position: Integer, nullable  -- (0, 1, 2, 3... dictating the strict render order)

        //-- Forwarding
        => is_forwarded: "Boolean, if the message was forwarded"

        //-- Audit & Sync
        => created_at: "Timestamp, Immutable"
        => updated_at: "Timestamp, Immutable"
        => hlc: "String (Hybrid Logical Clock), used for sync tracking"

    * MessageExclusions:
        => id: "UUID, Primary Key, Immutable"
        => message_root_id: "UUID, Foreign Key to Messages.root_id, Indexed, NOT NULL"
        => deleted_for: "UUID, Foreign Key to Users.id, Indexed, NOT NULL"
            // constraint: UNIQUE(message_root_id, deleted_for)
        => deleted_by: "UUID, Foreign Key to Users.id, Indexed, NOT NULL"

        //-- Audit & Sync
        => created_at: "Timestamp, Immutable"
        => updated_at: "Timestamp"
        => hlc: "String (Hybrid Logical Clock), used for sync tracking"

    * MessageReactions:
        => id: "UUID, Primary Key, Immutable" (Generated locally by WatermelonDB so users can message offline)

        //-- Foreign Keys
        => message_root_id: "UUID, Foreign Key to Messages.root_id, Indexed, NOT NULL"
        => message_id: "UUID, Foreign Key to Messages.id, Indexed, NOT NULL"
        => user_id: "UUID, Foreign Key to Users.id, Indexed, NOT NULL"

        //-- Constraint
        // UNIQUE (message_id, user_id)
        // This ensures a user can only have one active reaction per specific version.

        //-- Reaction Content
        => reaction: "String, NOT NULL (Unicode character) e.g., emoji or reaction type"
        => is_removed: "Boolean, NOT NULL, Default: false (The 'Tombstone' flag)"

        //-- Audit & Sync
        => created_at: "Timestamp, Immutable"
        => updated_at: "Timestamp, — changes on reaction update or removal for the specific version of the message"
        => hlc: "String (Hybrid Logical Clock), used for sync tracking"

    * Conversation:
        => id: "UUID, Primary Key, Immutable"

        //-- Conversation Type
        => type: "Enum ('direct', 'group', 'channel', 'system', 'self'), NOT NULL"

        //-- Shared Metadata based on conversation type
        => metadata<T extends 'group', 'channel', 'system'>: {

            => title: "String, NOT NULL",
            => description: "String, nullable",
            => avatar_url: "String, nullable",

            => is_allowed_invites: "Boolean, NOT NULL, DEFAULT false",
            => is_allowed_edit_info: "Boolean, NOT NULL, DEFAULT true",

            // Channel Metadata, when type = 'channel'
            => title: "String, NOT NULL",
            => description: "String, nullable",
            => avatar_url: "String, nullable",
            => created_by_id: "UUID, NOT NULL",
            => handle: "String, nullable, Unique",
            => is_allowed_messages: "Boolean, NOT NULL, DEFAULT true"

            // System Metadata, when type = 'system'
            => title: "String, NOT NULL",
            => description: "String, nullable",
            => avatar_url: "String, nullable"
        }

        //-- conversation level message pinning
        => pinned_message_root_id: "UUID, nullable"

        //-- Notification, personal message pinning and labeling

    * Message:
        => id: "UUID, Primary Key, Immutable" (Generated locally by WatermelonDB so users can message offline)
        => root_id: "UUID, Immutable, Indexed, NOT NULL (The constant ID for the whole "conversation thread" of this message)"
        => version: "Integer, Immutable, NOT NULL, Default: 0 (The version number of this message. 0-indexed: 0, 1, 2...)"
        //-- Unique Constraint Exception
        // Must enforce a UNIQUE index on (root_id, version)
        // to prevent duplicate version numbers within the same root_id.

        //-- Relational Links
        => conversation_id: "UUID, Foreign Key to Conversations.id, Indexed, NOT NULL"
        => sender_id: "UUID, Foreign Key to Users.id, Indexed, nullable"
           // (Nullable strictly for auto-generated system messages, e.g., 'Alice joined the group')
        => reply_to_message_id: "UUID, Foreign Key to Message.id, Indexed, nullable"
           // (Nullable for messages that are not replies to other messages). Snapshot pointer — the exact version that existed when the reply was sent.

        => type: "Enum ('text', 'image', 'video', 'audio', 'document', 'contact', 'location', 'poll', 'event', 'system', 'delete'), NOT NULL"

            => width: "Integer, width of the image in pixels, NOT NULL for type: 'image'"

            => height: "Integer, height of the video in pixels, NOT NULL for type: 'video'"
            => duration: "Integer, duration of the video in seconds, NOT NULL for type: 'video'"

            => size: "Integer, size of the document in bytes, NOT NULL for type: 'document'"
            => caption: "'Caption for the document', nullable for type: 'document'"

            => name: "'Name of the contact', nullable for type: 'contact'"
            => phone: "'Phone number of the contact', nullable for type: 'contact'"
            => email: "'Email address of the contact', nullable for type: 'contact'"
            => address: "'Address of the contact', nullable for type: 'contact'"
